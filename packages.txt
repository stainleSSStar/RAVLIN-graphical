# RAVLIN Graphical live ISO package list (installed on top of Void's base-system).
# One package per line; lines starting with # and blank lines are ignored.
#
# This is RAVLIN's packages.txt plus a minimal KDE Plasma 6 graphical stack.

# --- shell ---------------------------------------------------------------
# The default interactive shell for root and new users. The Void package is
# named "fish-shell"; the binary is /usr/bin/fish. Note that bash and dash
# stay installed: they come with base-system and most system scripts
# (including void-installer) rely on them.
fish-shell

# --- installer -----------------------------------------------------------
# TUI library used by void-installer, the interactive console installer.
dialog
# Mirror picker used by void-installer's Mirror menu (menu_mirror execs
# xmirror). Upstream's installer expects it on the live image; its local
# install path removes it from the target again.
xmirror

# --- drivers: wifi + LAN + GPU -------------------------------------------
# Full linux-firmware (not just linux-firmware-network): the graphical stack
# also needs GPU firmware blobs (amdgpu, i915, nouveau, radeon, ...).
# Ethernet and wifi drivers themselves live in the kernel; base-system also
# ships wifi-firmware, wpa_supplicant, iw and dhcpcd already.
linux-firmware

# --- drivers: sound -------------------------------------------------------
# ALSA mixer/tools, firmware for Sound Open Firmware devices (modern Intel
# laptops), and ALSA UCM profiles for correct routing on such hardware.
alsa-utils
sof-firmware
alsa-ucm-conf

# --- bootloader -----------------------------------------------------------
# void-installer installs GRUB to the target from the live environment
# (BIOS or EFI, including 32-bit EFI firmware found on older machines).
grub
grub-i386-efi
grub-x86_64-efi

# --- graphical stack: Xorg ------------------------------------------------
# xorg-minimal pulls xorg-server, xauth, xinit and xf86-input-libinput.
xorg-minimal
# Fallback video drivers for VMs and cards without KMS (QEMU std/vesa).
xf86-video-vesa
xf86-video-fbdev
# OpenGL drivers, including llvmpipe (software rendering) for VMs.
mesa-dri

# --- display manager ------------------------------------------------------
sddm

# --- KDE Plasma 6 (minimal session) ---------------------------------------
plasma-desktop
plasma-workspace
kwin
plasma-integration
breeze
breeze-icons
systemsettings
konsole
dolphin
plasma-pa
polkit-kde-agent
xdg-desktop-portal-kde

# --- audio server ----------------------------------------------------------
# Plasma 6 uses PipeWire; the pipewire package ships the PulseAudio
# compatibility layer (20-pipewire-pulse.conf), so no pulseaudio needed.
pipewire
wireplumber

# --- session management ----------------------------------------------------
# logind backend so Plasma's logout/shutdown/restart actions work on runit.
elogind
